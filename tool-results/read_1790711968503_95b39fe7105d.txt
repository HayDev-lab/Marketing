    26→// ===== types =====
    27→type ConnMode = "DIRECT_PUBLISH_AVAILABLE" | "DRAFT_TRANSFER_ONLY" | "USER_ACTION_REQUIRED" | "NOT_AVAILABLE";
    28→
    29→interface Connection {
    30→  id: string; platform: string; accountName: string | null; status: string; mode: ConnMode;
    31→}
    32→
    33→interface ScheduledPost {
    34→  id: string; platform: string; scheduledAt: string; timezone: string; status: string;
    35→  caption: string | null; preflightJson: string | null; error: string | null;
    36→  contentItem?: { id: string; title: string; approvalState: string } | null;
    37→}
    38→
    39→interface ContentLite { id: string; title: string; platform: string }
    40→
    41→interface AttemptResult { platform: string; caption: string | null; mediaUrl: string | null }
    42→
    43→const PLATFORM_TITLES: Record<string, string> = { instagram: "Instagram", tiktok: "TikTok", facebook: "Facebook", telegram: "Telegram" };
    44→const PLATFORMS = Object.keys(PLATFORM_TITLES);
    45→const TIMEZONES = ["Asia/Yerevan", "Europe/Moscow", "Europe/London", "America/New_York", "UTC"];
    46→
    47→function toLocalInput(d: Date): string {
    48→  const local = new Date(d.getTime() - d.getTimezoneOffset() * 60000);
    49→  return local.toISOString().slice(0, 16);
    50→}
    51→
    52→function postStatusAccent(status: string): { style?: React.CSSProperties; className?: string } {
    53→  switch (status) {
    54→    case "SCHEDULED":
    55→    case "PUBLISHED":
    56→      return { style: { background: "color-mix(in oklab, var(--neon-2) 16%, transparent)", color: "var(--neon-2)", borderColor: "color-mix(in oklab, var(--neon-2) 40%, transparent)" } };
    57→    case "READY":
    58→      return { style: { background: "color-mix(in oklab, var(--neon) 16%, transparent)", color: "var(--neon)", borderColor: "color-mix(in oklab, var(--neon) 40%, transparent)" } };
    59→    case "ACTION_REQUIRED":
    60→    case "SUBMITTING":
    61→    case "PROVIDER_ACCEPTED":
    62→      return { style: { background: "color-mix(in oklab, var(--neon-3) 18%, transparent)", color: "var(--neon-3)", borderColor: "color-mix(in oklab, var(--neon-3) 45%, transparent)" } };
    63→    case "FAILED":
    64→      return { className: "bg-destructive/15 text-destructive border-destructive/40" };
    65→    default:
    66→      return { className: "bg-muted/40 text-muted-foreground border-border" };
    67→  }
    68→}

    70→// ===== main module =====
    71→export function PublishingModule(_props: { onBrandsChanged?: () => void }) {
    72→  const { t, locale } = useI18n();
    73→
    74→  const [connections, setConnections] = useState<Connection[] | null>(null);
    75→  const [posts, setPosts] = useState<ScheduledPost[] | null>(null);
    76→  const [honestResult, setHonestResult] = useState<Record<string, { reason: string; requiredSetup: string }>>({});
    77→
    78→  // connect dialog
    79→  const [connectPlatform, setConnectPlatform] = useState<string | null>(null);
    80→  const [accountName, setAccountName] = useState("");
    81→  const [connecting, setConnecting] = useState(false);
    82→
    83→  // schedule dialog
    84→  const [schedOpen, setSchedOpen] = useState(false);
    85→  const [approvedItems, setApprovedItems] = useState<ContentLite[] | null>(null);
    86→  const [schedItemId, setSchedItemId] = useState<string>("");
    87→  const [schedPlatform, setSchedPlatform] = useState<string>("instagram");
    88→  const [schedAt, setSchedAt] = useState<string>(toLocalInput(new Date(Date.now() + 3600_000)));
    89→  const [schedTz, setSchedTz] = useState("Asia/Yerevan");
    90→  const [scheduling, setScheduling] = useState(false);
    91→  const [schedIssues, setSchedIssues] = useState<string[] | null>(null);
    92→
    93→  // per-post actions
    94→  const [attemptResult, setAttemptResult] = useState<AttemptResult | null>(null);
    95→  const [busyPost, setBusyPost] = useState<string | null>(null);
    96→  const [rescheduleId, setRescheduleId] = useState<string | null>(null);
    97→  const [rescheduleValue, setRescheduleValue] = useState("");
    98→
    99→  // week board
   100→  const [weekOffset, setWeekOffset] = useState(0);
   101→  const [weekPostId, setWeekPostId] = useState<string | null>(null);
   102→  const [weekReschedValue, setWeekReschedValue] = useState("");
   103→
   104→  const loadConnections = useCallback(async () => {
   105→    try {
   106→      setConnections(await api<Connection[]>("/api/publishing/connections"));
   107→    } catch {
   108→      setConnections([]);
   109→    }
   110→  }, []);
   111→
   112→  const loadPosts = useCallback(async () => {
   113→    try {
   114→      setPosts(await api<ScheduledPost[]>("/api/schedule"));
   115→    } catch {
   116→      setPosts([]);
   117→    }
   118→  }, []);
   119→
   120→  useEffect(() => { loadConnections(); loadPosts(); }, [loadConnections, loadPosts]);
   121→
   122→  const connect = async () => {
   123→    if (!connectPlatform) return;
   124→    setConnecting(true);
   125→    try {
   126→      const res = await api<{ connection: Connection; honestStatus: string; reason: string; requiredSetup: string }>(
   127→        "/api/publishing/connections",
   128→        { method: "POST", body: JSON.stringify({ platform: connectPlatform, accountName: accountName || undefined }) }
   129→      );
   130→      setConnections((prev) => {
   131→        const rest = (prev ?? []).filter((c) => c.platform !== res.connection.platform);
   132→        return [...rest, res.connection];
   133→      });
   134→      // Honest outcome: external OAuth credentials are absent — never pretend "Connected"
   135→      setHonestResult((prev) => ({ ...prev, [res.connection.platform]: { reason: res.reason, requiredSetup: res.requiredSetup } }));
   136→      toast.warning(t("publishing.connectBlocked"), { description: res.reason });
   137→      setConnectPlatform(null);
   138→      setAccountName("");
   139→    } catch (e) {
   140→      toast.error(t("publishing.err.default"), { description: (e as Error).message });
   141→    } finally {
   142→      setConnecting(false);
   143→    }
   144→  };
   145→
   146→  const disconnect = async (conn: Connection) => {
   147→    try {
   148→      await api("/api/publishing/connections", { method: "PATCH", body: JSON.stringify({ id: conn.id, disconnect: true }) });
   149→      setConnections((prev) => (prev ?? []).map((c) => (c.id === conn.id ? { ...c, status: "NOT_CONNECTED", accountName: null } : c)));
   150→      toast.success(`${PLATFORM_TITLES[conn.platform]} — ${t("publishing.status.NOT_CONNECTED")}`);
   151→    } catch (e) {
   152→      toast.error(t("publishing.err.default"), { description: (e as Error).message });
   153→    }
   154→  };
   155→
   156→  const openSchedule = async () => {
   157→    setSchedOpen(true);
   158→    setSchedIssues(null);
   159→    try {
   160→      const items = await api<ContentLite[]>("/api/content?state=APPROVED");
   161→      setApprovedItems(items);
   162→      if (items.length) {
   163→        setSchedItemId(items[0].id);
   164→        setSchedPlatform(items[0].platform);
   165→      } else {
   166→        setSchedItemId("");
   167→      }
   168→    } catch {
   169→      setApprovedItems([]);
   170→    }
   171→  };
   172→
   173→  const submitSchedule = async () => {
   174→    if (!schedItemId) return;
   175→    if (!schedAt || isNaN(new Date(schedAt).getTime())) {
   176→      toast.warning(t("publishing.dateTime"), { description: t("publishing.err.default") });
   177→      return;
   178→    }
   179→    setScheduling(true);
   180→    setSchedIssues(null);
   181→    try {
   182→      const res = await api<{ post: ScheduledPost; preflight: { ok: boolean; issues: string[] } }>(
   183→        "/api/schedule",
   184→        { method: "POST", body: JSON.stringify({ contentItemId: schedItemId, platform: schedPlatform, scheduledAt: new Date(schedAt).toISOString(), timezone: schedTz }) }
   185→      );
   186→      if (res.preflight.ok) {
   187→        toast.success(t("publishing.scheduleCreated"));
   188→        setSchedOpen(false);
   189→      } else {
   190→        setSchedIssues(res.preflight.issues);
   191→        toast.warning(t("publishing.scheduleWithIssues"), { description: res.preflight.issues.join(" · ") });
   192→      }
   193→      setPosts((prev) => (prev ? [res.post, ...prev] : [res.post]));
   194→    } catch (e) {
   195→      const code = (e as { code?: string }).code;
   196→      toast.error(code === "NOT_APPROVED" ? t("publishing.err.NOT_APPROVED") : t("publishing.err.default"), { description: (e as Error).message });
   197→    } finally {
   198→      setScheduling(false);
   199→    }
   200→  };
   201→
   202→  const cancelPost = async (post: ScheduledPost) => {
   203→    setBusyPost(post.id);
   204→    try {
   205→      const updated = await api<ScheduledPost>(`/api/schedule/${post.id}`, { method: "PATCH", body: JSON.stringify({ action: "cancel" }) });
   206→      setPosts((prev) => (prev ?? []).map((p) => (p.id === updated.id ? { ...p, ...updated } : p)));
   207→      toast.success(t("publishing.cancelled"));
   208→    } catch (e) {
   209→      toast.error(t("publishing.err.default"), { description: (e as Error).message });
   210→    } finally {
   211→      setBusyPost(null);
   212→    }
   213→  };
   214→
   215→  const reschedule = async (post: ScheduledPost, value: string) => {
   216→    if (!value || isNaN(new Date(value).getTime())) {
   217→      toast.warning(t("publishing.newDateTime"), { description: t("publishing.err.default") });
   218→      return;
   219→    }
   220→    setBusyPost(post.id);
   221→    try {
   222→      const updated = await api<ScheduledPost>(`/api/schedule/${post.id}`, { method: "PATCH", body: JSON.stringify({ action: "reschedule", scheduledAt: new Date(value).toISOString() }) });
   223→      setPosts((prev) => (prev ?? []).map((p) => (p.id === updated.id ? { ...p, ...updated } : p)));
   224→      setRescheduleId(null);
   225→      toast.success(t("publishing.rescheduled"));
   226→    } catch (e) {
   227→      toast.error(t("publishing.err.default"), { description: (e as Error).message });
   228→    } finally {
   229→      setBusyPost(null);
   230→    }
   231→  };
   232→
   233→  // Honest publish attempt: the API returns USER_ACTION_REQUIRED + export package (never a fake success)
   234→  const attemptPublish = async (post: ScheduledPost) => {
   235→    setBusyPost(post.id);
   236→    try {
   237→      const res = await api<{ post: ScheduledPost; honestStatus: string; exportPackage: { caption: string | null; platform: string; mediaUrl: string | null } }>(
   238→        `/api/schedule/${post.id}`,
   239→        { method: "PATCH", body: JSON.stringify({ action: "attempt_publish" }) }
   240→      );
   241→      setPosts((prev) => (prev ?? []).map((p) => (p.id === res.post.id ? { ...p, ...res.post } : p)));
   242→      setAttemptResult({ platform: res.exportPackage.platform, caption: res.exportPackage.caption, mediaUrl: res.exportPackage.mediaUrl });
   243→      toast.warning(t("publishing.attemptResult"), { description: t("publishing.attemptExplain") });
   244→    } catch (e) {
   245→      toast.error(t("publishing.err.default"), { description: (e as Error).message });
   246→    } finally {
   247→      setBusyPost(null);
   248→    }
   249→  };
   250→
   251→  const platformMeta = useMemo(() => {
   252→    const map: Record<string, Connection | undefined> = {};
   253→    for (const c of connections ?? []) map[c.platform] = c;
   254→    return map;
   255→  }, [connections]);
   256→
   257→  // week board data: Monday..Sunday of the selected week, local-date bucketed
   258→  const weekDays = useMemo(() => {
   259→    const now = new Date();
   260→    const monday = new Date(now.getFullYear(), now.getMonth(), now.getDate() - ((now.getDay() + 6) % 7) + weekOffset * 7);
   261→    return Array.from({ length: 7 }, (_, i) => new Date(monday.getFullYear(), monday.getMonth(), monday.getDate() + i));
   262→  }, [weekOffset]);
   263→  const dayKey = (d: Date) => `${d.getFullYear()}-${d.getMonth()}-${d.getDate()}`;
   264→  const postsByDay = useMemo(() => {
   265→    const map: Record<string, ScheduledPost[]> = {};
   266→    for (const p of posts ?? []) {
   267→      const k = dayKey(new Date(p.scheduledAt));
   268→      (map[k] ??= []).push(p);
   269→    }
   270→    for (const k of Object.keys(map)) map[k].sort((a, b) => +new Date(a.scheduledAt) - +new Date(b.scheduledAt));
   271→    return map;
   272→  }, [posts]);
   273→  const todayKey = dayKey(new Date());
   274→  const weekLabel = useMemo(() => {
   275→    const fmt = new Intl.DateTimeFormat(locale, { day: "numeric", month: "short" });
   276→    return `${fmt.format(weekDays[0])} — ${fmt.format(weekDays[6])}`;
   277→  }, [weekDays, locale]);
   278→  const weekdayFmt = useMemo(() => new Intl.DateTimeFormat(locale, { weekday: "short" }), [locale]);
   279→  const timeFmt = useMemo(() => new Intl.DateTimeFormat(locale, { hour: "2-digit", minute: "2-digit" }), [locale]);
   280→  const liveWeekPost = (posts ?? []).find((p) => p.id === weekPostId) ?? null;
   281→
   282→  return (
   283→    <div className="grid gap-5">
   284→      {/* header */}
   285→      <motion.section initial={{ opacity: 0, y: 12 }} animate={{ opacity: 1, y: 0 }} className="glass-strong neon-border rounded-2xl p-5 sm:p-6">
   286→        <div className="flex flex-wrap items-center justify-between gap-3">
   287→          <div className="min-w-0">
   288→            <h1 className="flex items-center gap-2 text-xl font-semibold sm:text-2xl">
   289→              <Send className="h-5 w-5 text-[var(--neon)]" aria-hidden /> {t("publishing.title")}
   290→            </h1>
   291→            <p className="mt-1 text-sm text-muted-foreground">{t("publishing.connectionsDesc")}</p>
   292→          </div>
   293→          <Button size="sm" className="min-h-11" onClick={openSchedule} aria-label={t("publishing.scheduleNew")}>
   294→            <CalendarClock className="h-4 w-4" aria-hidden /> {t("publishing.scheduleNew")}
   295→          </Button>
   296→        </div>
   297→      </motion.section>
   298→
   299→      {/* connections */}
   300→      <section aria-label={t("publishing.connections")}>
   301→        <h2 className="mb-3 text-sm font-medium tracking-wide text-muted-foreground">{t("publishing.connections")}</h2>
   302→        {!connections ? (
   303→          <div className="grid gap-3 sm:grid-cols-2 xl:grid-cols-4">{[...Array(4)].map((_, i) => <Skeleton key={i} className="h-44 rounded-2xl bg-muted/40" />)}</div>
   304→        ) : (
   305→          <div className="grid gap-3 sm:grid-cols-2 xl:grid-cols-4">
   306→            {PLATFORMS.map((p, i) => {
   307→              const conn = platformMeta[p];
   308→              const honest = honestResult[p];
   309→              return (
   310→                <motion.div key={p} initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: i * 0.05 }} className="glass glass-hover flex flex-col gap-3 rounded-2xl p-4">
   311→                  <div className="flex items-center justify-between gap-2">
   312→                    <span className="flex items-center gap-2 text-sm font-semibold">
   313→                      <span className="rounded-lg p-1.5" style={{ background: "color-mix(in oklab, var(--neon) 14%, transparent)" }}>
   314→                        <PlatformIcon platform={p} className="h-4 w-4 text-[var(--neon)]" />
   315→                      </span>
   316→                      {PLATFORM_TITLES[p]}
   317→                    </span>
   318→                    <Badge variant="outline" className="shrink-0 text-[10px]">{t(`publishing.mode.${conn?.mode ?? "NOT_AVAILABLE"}`)}</Badge>
   319→                  </div>
   320→                  <p className="text-xs leading-relaxed text-muted-foreground">{t(`publishing.mode.explain.${conn?.mode ?? "NOT_AVAILABLE"}`)}</p>
   321→                  <div className="mt-auto grid gap-2">
   322→                    <div className="flex items-center justify-between text-[11px] text-muted-foreground">
   323→                      <span>{t("common.status")}: {conn?.status === "NOT_CONNECTED" ? t("publishing.status.NOT_CONNECTED") : conn?.status === "CONNECTED" ? t("publishing.status.CONNECTED") : t("publishing.status.ACTION_REQUIRED")}</span>
   324→                      {conn?.accountName && <span className="truncate max-w-24">{conn.accountName}</span>}
   325→                    </div>
   326→                    {conn && !conn.id.startsWith("virtual_") ? (
   327→                      <Button size="sm" variant="outline" className="min-h-11" onClick={() => disconnect(conn)} aria-label={t("publishing.disconnect")}>
   328→                        <Unlink className="h-4 w-4" aria-hidden /> {t("publishing.disconnect")}
   329→                      </Button>
   330→                    ) : (
   331→                      <Button size="sm" variant="outline" className="min-h-11" onClick={() => { setConnectPlatform(p); setAccountName(""); }} aria-label={`${t("publishing.connect")} ${PLATFORM_TITLES[p]}`}>
   332→                        <Link2 className="h-4 w-4" aria-hidden /> {t("publishing.connect")}
   333→                      </Button>
   334→                    )}
   335→                  </div>
   336→                  {honest && (
   337→                    <div className="rounded-xl border border-[color-mix(in_oklab,var(--neon-3)_50%,transparent)] bg-[color-mix(in_oklab,var(--neon-3)_10%,transparent)] p-3 text-[11px]" role="alert">
   338→                      <p className="flex items-center gap-1 font-medium text-[var(--neon-3)]">
   339→                        <TriangleAlert className="h-3.5 w-3.5" aria-hidden /> {t("publishing.connectBlocked")}
   340→                      </p>
   341→                      <p className="mt-1 text-foreground/80">{honest.reason}</p>
   342→                      <p className="mt-1 text-muted-foreground"><span className="font-medium">{t("publishing.requiredSetup")}:</span> {honest.requiredSetup}</p>
   343→                    </div>
   344→                  )}
   345→                </motion.div>
   346→              );
   347→            })}
   348→          </div>
   349→        )}
   350→      </section>
   351→
   352→      {/* honest publish attempt result */}
   353→      {attemptResult && (
   354→        <motion.div initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} role="alert">
   355→          <Card className="rounded-2xl border-[color-mix(in_oklab,var(--neon-3)_50%,transparent)] bg-[color-mix(in_oklab,var(--neon-3)_8%,transparent)]">
   356→            <CardHeader className="pb-2">
   357→              <CardTitle className="flex items-center gap-2 text-base">
   358→                <Hand className="h-4 w-4 text-[var(--neon-3)]" aria-hidden /> {t("publishing.attemptResult")}
   359→              </CardTitle>
   360→            </CardHeader>
   361→            <CardContent className="grid gap-3 p-4 pt-0 text-sm">
   362→              <p className="text-foreground/85">{t("publishing.attemptExplain")}</p>
   363→              {attemptResult.caption && (
   364→                <>
   365→                  <p className="text-xs font-medium text-muted-foreground">{t("publishing.exportCaption")}</p>
   366→                  <pre className="max-h-32 overflow-y-auto whitespace-pre-wrap rounded-xl border border-border/60 bg-background/40 p-3 text-xs scrollbar-thin">{attemptResult.caption}</pre>
   367→                </>
   368→              )}
   369→              {attemptResult.mediaUrl ? (
   370→                <a href={attemptResult.mediaUrl} download className="inline-flex min-h-11 w-fit items-center gap-2 rounded-xl border border-border/60 px-4 text-sm font-medium glass-hover" aria-label={t("publishing.exportMedia")}>
   371→                  <Download className="h-4 w-4 text-[var(--neon-2)]" aria-hidden /> {t("publishing.exportMedia")}
   372→                </a>
   373→              ) : (
   374→                <p className="text-xs text-muted-foreground">{t("publishing.noMediaToExport")}</p>
   375→              )}
   376→              <Button size="sm" variant="ghost" className="min-h-11 w-fit" onClick={() => setAttemptResult(null)}>{t("common.close")}</Button>
   377→            </CardContent>
   378→          </Card>
   379→        </motion.div>
   380→      )}
   381→
   382→      {/* week board */}
   383→      <Card className="glass rounded-2xl">
   384→        <CardHeader className="flex-row flex-wrap items-center justify-between gap-2 pb-2">
   385→          <CardTitle className="flex items-center gap-2 text-base">
   386→            <CalendarRange className="h-4 w-4 text-[var(--neon)]" aria-hidden /> {t("publishing.weekBoard")}
   387→          </CardTitle>
   388→          <div className="flex items-center gap-1.5">
   389→            <Button variant="outline" size="icon" className="h-9 w-9" aria-label={t("publishing.weekPrev")} onClick={() => setWeekOffset((w) => w - 1)}>
   390→              <ChevronLeft className="h-4 w-4" aria-hidden />
   391→            </Button>
   392→            <span className="min-w-[118px] text-center text-xs font-medium text-muted-foreground" aria-live="polite">{weekLabel}</span>
   393→            <Button variant="outline" size="icon" className="h-9 w-9" aria-label={t("publishing.weekNext")} onClick={() => setWeekOffset((w) => w + 1)}>
   394→              <ChevronRight className="h-4 w-4" aria-hidden />
   395→            </Button>
   396→            {weekOffset !== 0 && (
   397→              <Button variant="ghost" size="sm" className="min-h-9 text-xs" onClick={() => setWeekOffset(0)}>{t("publishing.thisWeek")}</Button>
   398→            )}
   399→          </div>
   400→        </CardHeader>
   401→        <CardContent className="grid gap-2">
   402→          <p className="text-xs text-muted-foreground">{t("publishing.weekHint")}</p>
   403→          <div
   404→            className="grid grid-flow-col auto-cols-[minmax(138px,1fr)] gap-2 overflow-x-auto pb-1 scrollbar-thin sm:auto-cols-fr sm:grid-flow-row sm:grid-cols-7"
   405→            role="table"
   406→            aria-label={t("publishing.weekBoard")}
   407→          >
   408→            {weekDays.map((d) => {
   409→              const k = dayKey(d);
   410→              const dayPosts = postsByDay[k] ?? [];
   411→              const isToday = k === todayKey;
   412→              return (
