    22→const LAST_PROJECT_KEY = "haydev-video-last";

    24→
    25→interface Scene {
    26→  id: string; order: number; durationSec: number; prompt: string; narration?: string | null;
    27→  status: string; assetId?: string | null; jobId?: string | null; error?: string | null; cost?: number | null; version: number;
    28→}
    29→interface Project {
    30→  id: string; title: string; durationSec: number; aspectRatio: string; language: string;
    31→  script?: string | null; characterBibleJson?: string | null; styleBibleJson?: string | null;
    32→  status: string; createdAt: string; updatedAt: string; scenes?: Scene[]; _count?: { scenes: number };
    33→}
    34→interface JobInfo { id: string; status: string; error?: string | null }
    35→
    36→type BibleRows = { key: string; value: string }[];
    37→
    38→function parseBible(raw: string | null | undefined): BibleRows {
    39→  if (!raw) return [];
    40→  try {
    41→    const obj = JSON.parse(raw) as Record<string, unknown>;
    42→    return Object.entries(obj).map(([key, value]) => ({ key, value: String(value ?? "") }));
    43→  } catch {
    44→    return [];
    45→  }
    46→}
    47→
    48→function bibleToObj(rows: BibleRows): Record<string, string> {
    49→  const obj: Record<string, string> = {};
    50→  for (const r of rows) if (r.key.trim()) obj[r.key.trim()] = r.value;
    51→  return obj;
    52→}

    56→  { id: "9:16", w: 20, h: 36 },
    57→  { id: "1:1", w: 28, h: 28 },
    58→  { id: "16:9", w: 48, h: 27 },

    61→export function VideoStudioModule() {
    62→  const { t, localeLabels } = useI18n();
    63→  const activeBrandId = useApp((s) => s.activeBrandId);
    64→
    65→  const [projects, setProjects] = useState<Project[]>([]);
    66→  const [loadingList, setLoadingList] = useState(true);
    67→  const [view, setView] = useState<"list" | "editor">("list");
    68→  const [project, setProject] = useState<Project | null>(null);
    69→  const projectRef = useRef<Project | null>(null);
    70→  projectRef.current = project;
    71→
    72→  // create form
    73→  const [title, setTitle] = useState("");
    74→  const [durationSec, setDurationSec] = useState(30);
    75→  const [aspect, setAspect] = useState("9:16");
    76→  const [language, setLanguage] = useState("hy");
    77→  const [brief, setBrief] = useState("");
    78→  const [creating, setCreating] = useState(false);
    79→
    80→  // editor state
    81→  const [scriptBrief, setScriptBrief] = useState("");
    82→  const [scripting, setScripting] = useState(false);
    83→  const [promptDrafts, setPromptDrafts] = useState<Record<string, string>>({});
    84→  const [savingPrompt, setSavingPrompt] = useState<string | null>(null);
    85→  const [sceneBusy, setSceneBusy] = useState<string | null>(null);
    86→  const [polls, setPolls] = useState<Record<string, { jobId: string; elapsed: number }>>({});
    87→  const pollsRef = useRef(polls);
    88→  pollsRef.current = polls;
    89→  const [charRows, setCharRows] = useState<BibleRows>([]);
    90→  const [styleRows, setStyleRows] = useState<BibleRows>([]);
    91→  const [savingBible, setSavingBible] = useState<"char" | "style" | null>(null);
    92→
    93→  const loadList = useCallback(async () => {
    94→    try {
    95→      const list = await api<Project[]>("/api/video-projects");
    96→      setProjects(list);
    97→    } catch {
    98→      /* list stays empty */
    99→    } finally {
   100→      setLoadingList(false);
   101→    }
   102→  }, []);
   103→
   104→  const refreshProject = useCallback(async () => {
   105→    const id = projectRef.current?.id;
   106→    if (!id) return;
   107→    try {
   108→      const fresh = await api<Project>(`/api/video-projects?id=${id}`);
   109→      setProject(fresh);
   110→      loadList();
   111→    } catch {
   112→      /* keep current */
   113→    }
   114→  }, [loadList]);
   115→
   116→  const startPoll = useCallback((sceneId: string, jobId: string) => {
   117→    setPolls((prev) => (prev[sceneId] ? prev : { ...prev, [sceneId]: { jobId, elapsed: 0 } }));
   118→  }, []);
   119→
   120→  const openProject = useCallback(async (id: string) => {
   121→    try {
   122→      const p = await api<Project>(`/api/video-projects?id=${id}`);
   123→      setProject(p);
   124→      setScriptBrief("");
   125→      setPromptDrafts({});
   126→      setCharRows(parseBible(p.characterBibleJson));
   127→      setStyleRows(parseBible(p.styleBibleJson));
   128→      setView("editor");
   129→      try {
   130→        window.localStorage.setItem(LAST_PROJECT_KEY, id);
   131→      } catch { /* non-critical */ }
   132→      for (const sc of p.scenes ?? []) {
   133→        if (sc.jobId && ["GENERATING", "QUEUED"].includes(sc.status)) startPoll(sc.id, sc.jobId);
   134→      }
   135→    } catch (err) {
   136→      showStudioError(t, err);
   137→    }
   138→  }, [startPoll]);
   139→
   140→  // hydrate: list + last opened project + library hand-off
   141→  useEffect(() => {
   142→    (async () => {
   143→      await loadList();
   144→      try {
   145→        const fromLib = window.localStorage.getItem(BRIEF_FROM_LIB_KEY);
   146→        if (fromLib) {
   147→          setBrief(fromLib);
   148→          window.localStorage.removeItem(BRIEF_FROM_LIB_KEY);
   149→          toast.info(t("studio.vid.briefLoaded"));
   150→        }
   151→        const last = window.localStorage.getItem(LAST_PROJECT_KEY);
   152→        if (last) {
   153→          const exists = await api<Project[]>(`/api/video-projects`).catch(() => []);
   154→          if (exists.some((p) => p.id === last)) await openProject(last);
   155→        }
   156→      } catch { /* non-critical */ }
   157→    })();
   158→  }, []);
   159→
   160→  // job polling: reconcile every 6s while scenes are in flight
   161→  useEffect(() => {
   162→    const iv = setInterval(async () => {
   163→      const entries = Object.entries(pollsRef.current);
   164→      for (const [sceneId, p] of entries) {
   165→        try {
   166→          const job = await api<JobInfo>(`/api/jobs/${p.jobId}`);
   167→          if (job.status === "COMPLETED") {
   168→            setPolls((prev) => { const n = { ...prev }; delete n[sceneId]; return n; });
   169→            await refreshProject();
   170→            pulseCore("SUCCESS");
   171→            toast.success(t("studio.vid.status.COMPLETED"));
   172→          } else if (["FAILED", "CANCELLED"].includes(job.status)) {
   173→            setPolls((prev) => { const n = { ...prev }; delete n[sceneId]; return n; });
   174→            await refreshProject();
   175→            pulseCore("ERROR");
   176→            toast.error(t("studio.err.generic", { msg: job.error ?? "video generation failed" }));
   177→          }
   178→        } catch { /* transient network/provider error — keep polling */ }
   179→      }
   180→    }, 6000);
   181→    return () => clearInterval(iv);
   182→  }, [refreshProject, t]);
   183→
   184→  // elapsed counter for polling display
   185→  useEffect(() => {
   186→    const iv = setInterval(() => {
   187→      setPolls((prev) => {
   188→        const next: Record<string, { jobId: string; elapsed: number }> = {};
   189→        for (const [k, v] of Object.entries(prev)) next[k] = { ...v, elapsed: v.elapsed + 1 };
   190→        return next;
   191→      });
   192→    }, 1000);
   193→    return () => clearInterval(iv);
   194→  }, []);
   195→
   196→  const backToList = () => {
   197→    setView("list");
   198→    setProject(null);
   199→    loadList();
   200→  };
   201→
   202→  const createProject = async () => {
   203→    if (!title.trim()) {
   204→      toast.warning(t("studio.vid.projectTitle"));
   205→      return;
   206→    }
   207→    setCreating(true);
   208→    pulseCore("PLANNING");
   209→    try {
   210→      const p = await api<Project>("/api/generate/video", {
   211→        method: "POST",
   212→        body: JSON.stringify({
   213→          action: "create_project",
   214→          brandId: activeBrandId ?? undefined,
   215→          title,
   216→          durationSec,
   217→          aspectRatio: aspect,
   218→          language,
   219→          script: brief.trim() || undefined,
   220→          autoShotPlan: true,
   221→        }),
   222→      });
   223→      setTitle("");
   224→      setBrief("");
   225→      toast.success(t("studio.vid.created"));
   226→      await loadList();
   227→      await openProject(p.id);
   228→    } catch (err) {
   229→      pulseCore("ERROR");
   230→      showStudioError(t, err);
   231→    } finally {
   232→      setCreating(false);
   233→    }
   234→  };
   235→
   236→  const generateScript = async () => {
   237→    if (!project) return;
   238→    setScripting(true);
   239→    pulseCore("ANALYZING");
   240→    try {
   241→      const updated = await api<Project>("/api/generate/video", {
   242→        method: "POST",
   243→        body: JSON.stringify({ action: "generate_script", projectId: project.id, brief: scriptBrief || undefined }),
   244→      });
   245→      setProject((prev) => (prev ? { ...prev, ...updated, scenes: prev.scenes } : prev));
   246→      setCharRows(parseBible(updated.characterBibleJson));
   247→      setStyleRows(parseBible(updated.styleBibleJson));
   248→      pulseCore("SUCCESS");
   249→      toast.success(t("common.success"));
   250→      loadList();
   251→    } catch (err) {
   252→      pulseCore("ERROR");
   253→      showStudioError(t, err);
   254→    } finally {
   255→      setScripting(false);
   256→    }
   257→  };
   258→
   259→  const saveScenePrompt = async (sceneId: string) => {
   260→    if (!project) return;
   261→    const prompt = promptDrafts[sceneId];
   262→    if (prompt === undefined) return;
   263→    setSavingPrompt(sceneId);
   264→    try {
   265→      await api<Project>("/api/video-projects", {
   266→        method: "PATCH",
   267→        body: JSON.stringify({ projectId: project.id, sceneId, prompt }),
   268→      });
   269→      setProject((prev) =>
   270→        prev ? { ...prev, scenes: prev.scenes?.map((s) => (s.id === sceneId ? { ...s, prompt } : s)) } : prev
   271→      );
   272→      toast.success(t("studio.vid.promptSaved"));
   273→    } catch (err) {
   274→      showStudioError(t, err);
   275→    } finally {
   276→      setSavingPrompt(null);
   277→    }
   278→  };
   279→
   280→  const generateScene = async (scene: Scene) => {
   281→    if (!project) return;
   282→    setSceneBusy(scene.id);
   283→    try {
   284→      if (scene.status === "COMPLETED" && scene.assetId) {
   285→        // replace: reset scene first (version+1), then submit a fresh job
   286→        await api<Project>("/api/video-projects", {
   287→          method: "PATCH",
   288→          body: JSON.stringify({ projectId: project.id, sceneId: scene.id, reset: true }),
   289→        });
   290→      }
   291→      pulseCore("GENERATING");
   292→      const res = await api<{ jobId: string; assetId?: string; deduplicated?: boolean }>("/api/generate/video", {
   293→        method: "POST",
   294→        body: JSON.stringify({ action: "generate_scene", sceneId: scene.id }),
   295→      });
   296→      if (res.deduplicated && res.assetId) {
   297→        await refreshProject();
   298→        pulseCore("SUCCESS");
   299→        toast.success(t("studio.vid.status.COMPLETED"));
   300→      } else if (res.jobId) {
   301→        startPoll(scene.id, res.jobId);
   302→        await refreshProject();
   303→      }
   304→    } catch (err) {
   305→      pulseCore("ERROR");
   306→      showStudioError(t, err);
   307→    } finally {
   308→      setSceneBusy(null);
   309→    }
   310→  };
   311→
   312→  const jobAction = async (scene: Scene, action: "resume" | "cancel") => {
   313→    if (!scene.jobId) return;
   314→    setSceneBusy(scene.id);
   315→    try {
   316→      if (action === "resume") {
   317→        pulseCore("GENERATING");
   318→        await api<JobInfo>(`/api/jobs/${scene.jobId}`, { method: "POST", body: JSON.stringify({ action: "resume" }) });
   319→        startPoll(scene.id, scene.jobId);
   320→        await refreshProject();
   321→      } else {
   322→        await api<JobInfo>(`/api/jobs/${scene.jobId}`, { method: "POST", body: JSON.stringify({ action: "cancel" }) });
   323→        await refreshProject();
   324→        toast.info(t("common.cancel"));
   325→      }
   326→    } catch (err) {
   327→      showStudioError(t, err);
   328→    } finally {
   329→      setSceneBusy(null);
   330→    }
   331→  };
   332→
   333→  const saveBible = async (kind: "char" | "style") => {
   334→    if (!project) return;
   335→    setSavingBible(kind);
   336→    try {
   337→      const body = kind === "char"
   338→        ? { projectId: project.id, characterBible: bibleToObj(charRows) }
   339→        : { projectId: project.id, styleBible: bibleToObj(styleRows) };
   340→      await api<Project>("/api/video-projects", { method: "PATCH", body: JSON.stringify(body) });
   341→      toast.success(t("studio.vid.bibleSaved"));
   342→    } catch (err) {
   343→      showStudioError(t, err);
   344→    } finally {
   345→      setSavingBible(null);
   346→    }
   347→  };
   348→
   349→  const scenes = project?.scenes ?? [];
   350→  const estTotal = scenes.reduce((sum, s) => sum + (s.cost ?? (s.status === "COMPLETED" ? 0 : 0.1)), 0);
   351→
   352→  const statusBadge = (status: string) => {
   353→    const map: Record<string, string> = {
   354→      COMPLETED: "border-[var(--neon-2)]/50 text-[var(--neon-2)]",
   355→      GENERATING: "border-[var(--neon)]/50 text-[var(--neon)]",
   356→      FAILED: "border-destructive/50 text-destructive",
   357→    };
   358→    return (
   359→      <Badge variant="outline" className={`shrink-0 text-[10px] ${map[status] ?? "text-muted-foreground"}`}>
   360→        {t(`studio.vid.status.${status}` as const)}
   361→      </Badge>
   362→    );
   363→  };
   364→
   365→  return (
   366→    <div className="grid gap-5">
   367→      <motion.section initial={{ opacity: 0, y: 12 }} animate={{ opacity: 1, y: 0 }} className="glass-strong neon-border rounded-2xl p-5 sm:p-6">
   368→        <div className="flex flex-wrap items-center justify-between gap-3">
   369→          <div>
   370→            <h1 className="text-xl font-semibold sm:text-2xl neon-text">{t("studio.vid.title")}</h1>
   371→            <p className="mt-1 text-sm text-muted-foreground">{t("studio.vid.subtitle")}</p>
   372→          </div>
   373→          {view === "editor" && (
   374→            <Button variant="outline" className="min-h-11 gap-2" onClick={backToList}>
   375→              <ArrowLeft className="h-4 w-4" /> {t("studio.vid.back")}
   376→            </Button>
   377→          )}
   378→        </div>
   379→      </motion.section>
   380→
   381→      {view === "list" ? (
   382→        <>
   383→          {/* create form */}
   384→          <Card className="glass-strong neon-border rounded-2xl">
   385→            <CardHeader className="pb-3">
   386→              <CardTitle className="flex items-center gap-2 text-base">
   387→                <Plus className="h-4 w-4 text-[var(--neon)]" /> {t("studio.vid.newProject")}
   388→              </CardTitle>
   389→            </CardHeader>
   390→            <CardContent className="grid gap-5">
   391→              <div className="grid gap-2 sm:grid-cols-2">
   392→                <div className="grid gap-2">
   393→                  <Label htmlFor="vid-title">{t("studio.vid.projectTitle")}</Label>
   394→                  <Input id="vid-title" value={title} onChange={(e) => setTitle(e.target.value)} placeholder={t("studio.vid.titlePh")} />
   395→                </div>
   396→                <div className="grid gap-2">
   397→                  <Label htmlFor="vid-lang">{t("studio.vid.language")}</Label>
   398→                  <Select value={language} onValueChange={setLanguage}>
   399→                    <SelectTrigger id="vid-lang" className="min-h-11 w-full"><SelectValue /></SelectTrigger>
   400→                    <SelectContent>
   401→                      {Object.entries(localeLabels).map(([id, label]) => (
   402→                        <SelectItem key={id} value={id}>{label}</SelectItem>
   403→                      ))}
   404→                    </SelectContent>
   405→                  </Select>
   406→                </div>
   407→              </div>
   408→
   409→              <div className="grid gap-2">
   410→                <Label>{t("studio.vid.duration")}</Label>
   411→                <div className="grid grid-cols-3 gap-2" role="radiogroup" aria-label={t("studio.vid.duration")}>
   412→                  {DURATIONS.map((d) => (
   413→                    <button
   414→                      key={d}
   415→                      role="radio"
   416→                      aria-checked={durationSec === d}
   417→                      onClick={() => setDurationSec(d)}
   418→                      className={`glass glass-hover min-h-14 rounded-xl text-center transition ${durationSec === d ? "neon-border neon-text" : "border border-border/60"}`}
   419→                    >
   420→                      <span className="text-lg font-bold">{d}s</span>
   421→                    </button>
   422→                  ))}
   423→                </div>
   424→              </div>
   425→
   426→              <div className="grid gap-2">
   427→                <Label>{t("studio.vid.aspect")}</Label>
   428→                <div className="flex flex-wrap gap-2" role="radiogroup" aria-label={t("studio.vid.aspect")}>
   429→                  {ASPECTS.map((a) => (
   430→                    <button
   431→                      key={a.id}
   432→                      role="radio"
   433→                      aria-checked={aspect === a.id}
   434→                      onClick={() => setAspect(a.id)}
   435→                      className={`glass glass-hover flex min-h-11 min-w-11 flex-col items-center justify-center gap-1.5 rounded-xl px-3 py-2 text-xs transition ${aspect === a.id ? "neon-border neon-text" : "border border-border/60"}`}
   436→                    >
   437→                      <span
   438→                        className={`rounded-[4px] border-2 ${aspect === a.id ? "border-[var(--neon)] bg-[var(--neon)]/15" : "border-muted-foreground/50"}`}
   439→                        style={{ width: a.w, height: a.h }}
   440→                        aria-hidden
   441→                      />
   442→                      <span className="font-mono">{a.id}</span>
   443→                    </button>
   444→                  ))}
   445→                </div>
   446→              </div>
   447→
   448→              <div className="grid gap-2">
   449→                <Label htmlFor="vid-brief">{t("studio.vid.brief")}</Label>
   450→                <Textarea id="vid-brief" value={brief} onChange={(e) => setBrief(e.target.value)} placeholder={t("studio.vid.briefPh")} rows={3} className="resize-none" />
   451→                <p className="text-[11px] text-muted-foreground">
   452→                  {t("studio.vid.brand")}: {activeBrandId ? <span className="neon-text font-medium">{activeBrandId.slice(0, 14)}…</span> : t("studio.vid.brandNone")}
   453→                </p>
   454→              </div>
   455→
   456→              <Button onClick={createProject} disabled={creating} size="lg" className="min-h-11 gap-2 font-semibold">
   457→                {creating ? <Loader2 className="h-4 w-4 animate-spin" /> : <Wand2 className="h-4 w-4" />}
   458→                {creating ? t("studio.vid.creating") : t("studio.vid.create")}
   459→              </Button>
   460→            </CardContent>
   461→          </Card>
   462→
   463→          {/* project list */}
   464→          <Card className="glass rounded-2xl">
   465→            <CardHeader className="pb-3">
   466→              <CardTitle className="flex items-center gap-2 text-base">
   467→                <Film className="h-4 w-4 text-[var(--neon-3)]" /> {t("studio.vid.projects")}
   468→              </CardTitle>
   469→            </CardHeader>
   470→            <CardContent>
   471→              {loadingList ? (
   472→                <div className="grid gap-2">{[...Array(3)].map((_, i) => <Skeleton key={i} className="h-16 rounded-xl bg-muted/40" />)}</div>
   473→              ) : projects.length === 0 ? (
   474→                <p className="py-6 text-center text-sm text-muted-foreground">{t("studio.vid.projectsEmpty")}</p>
   475→              ) : (
   476→                <ul className="grid max-h-96 gap-2 overflow-y-auto scrollbar-thin">
   477→                  {projects.map((p) => (
   478→                    <li key={p.id} className="glass glass-hover flex flex-wrap items-center justify-between gap-3 rounded-xl p-3">
   479→                      <div className="min-w-0 flex-1">
   480→                        <p className="truncate text-sm font-medium">{p.title}</p>
   481→                        <p className="mt-0.5 text-[11px] text-muted-foreground">
   482→                          {p.durationSec}s · {p.aspectRatio} · {p.language.toUpperCase()} ·{" "}
   483→                          {t("studio.vid.sceneCount", { n: p._count?.scenes ?? 0 })}
   484→                        </p>
   485→                      </div>
   486→                      <div className="flex items-center gap-2">
   487→                        {statusBadge(p.status)}
   488→                        <Button variant="outline" size="sm" className="min-h-11 gap-1.5 text-xs" onClick={() => openProject(p.id)}>
   489→                          <Play className="h-3.5 w-3.5 text-[var(--neon)]" /> {t("studio.vid.open")}
   490→                        </Button>
   491→                      </div>
   492→                    </li>
   493→                  ))}
   494→                </ul>
   495→              )}
   496→            </CardContent>
   497→          </Card>
   498→        </>
