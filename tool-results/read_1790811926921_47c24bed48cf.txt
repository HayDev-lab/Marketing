    54→function parseProjectMeta(raw: string | null | undefined): { soundtrack?: { musicAssetId: string; loopOverride: boolean }; voiceover?: VoiceoverMeta | null; assembly?: Record<string, unknown> } {
    55→  try { return raw ? (JSON.parse(raw) as Record<string, never>) : {}; } catch { return {}; }
    56→}

    58→/** Default narration text: scene narrations (in order) → full script → empty. */
    59→function defaultVoiceText(p: Project | null): string {
    60→  if (!p) return "";
    61→  const narrations = (p.scenes ?? []).map((s) => (s.narration ?? "").trim()).filter(Boolean);
    62→  if (narrations.length) return narrations.join("\n\n");
    63→  return p.script?.trim() ?? "";
    64→}
    65→
    66→type BibleRows = { key: string; value: string }[];
    67→
    68→function parseBible(raw: string | null | undefined): BibleRows {
    69→  if (!raw) return [];
    70→  try {
    71→    const obj = JSON.parse(raw) as Record<string, unknown>;
    72→    return Object.entries(obj).map(([key, value]) => ({ key, value: String(value ?? "") }));
    73→  } catch {
    74→    return [];
    75→  }
    76→}
    77→
    78→function bibleToObj(rows: BibleRows): Record<string, string> {
    79→  const obj: Record<string, string> = {};
    80→  for (const r of rows) if (r.key.trim()) obj[r.key.trim()] = r.value;
    81→  return obj;
    82→}
