    63→function fmtSize(bytes: number): string {
    64→  return bytes >= 1024 * 1024 ? `${(bytes / 1024 / 1024).toFixed(1)} MB` : `${Math.max(1, Math.round(bytes / 1024))} KB`;

    67→// ---------- Web Audio preview: apply edit params to a decoded buffer (real DSP) ----------
    68→
    69→function renderPreviewBuffer(ctx: AudioContext, src: AudioBuffer, edit: EditState): AudioBuffer {
    70→  const sr = src.sampleRate;
    71→  const trimStart = Math.min(edit.trimStartSec, Math.max(0, src.duration - 0.2));
    72→  const trimEnd = edit.trimEndSec !== null ? Math.min(edit.trimEndSec, src.duration) : src.duration;
    73→  const len = Math.max(1, Math.floor((trimEnd - trimStart) * sr));
    74→  const out = ctx.createBuffer(2, len, sr);
    75→  const fadeInN = Math.min(len, Math.floor(edit.fadeInSec * sr));
    76→  const fadeOutN = Math.min(len, Math.floor(edit.fadeOutSec * sr));
    77→  for (let ch = 0; ch < 2; ch++) {
    78→    const inData = src.getChannelData(Math.min(ch, src.numberOfChannels - 1));
    79→    const outData = out.getChannelData(ch);
    80→    for (let i = 0; i < len; i++) {
    81→      let g = edit.volume;
    82→      if (i < fadeInN) g *= i / fadeInN;
    83→      if (i >= len - fadeOutN) g *= (len - i) / fadeOutN;
    84→      outData[i] = inData[trimStart * sr + i] * g;
    85→    }
    86→  }
    87→  return out;
    88→}
