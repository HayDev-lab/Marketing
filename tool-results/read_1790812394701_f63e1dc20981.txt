   335→function ProvidersTab() {

   343→    } catch {
   344→      setProviders([]);
   345→    }
   346→  }, []);
   347→
   348→  useEffect(() => { load(); }, [load]);
   349→
   350→  const toggle = async (p: Provider, enabled: boolean) => {
   351→    try {
   352→      await api("/api/settings/providers", { method: "PATCH", body: JSON.stringify({ providerId: p.providerId, enabled }) });
   353→      setProviders((prev) => (prev ?? []).map((x) => (x.providerId === p.providerId ? { ...x, enabled } : x)));
   354→      toast.success(`${p.title}: ${enabled ? t("common.approve") : t("settings.status.DISABLED")}`);
   355→    } catch (e) {
   356→      // 423 BLOCKED_EXTERNAL — honest refusal, provider stays disabled
   357→      toast.error(t("settings.blockedNote"), { description: (e as Error).message });
   358→    }
   359→  };
   360→
   361→  const healthCheck = async (p: Provider) => {
   362→    setChecking(p.providerId);
   363→    try {
   364→      const res = await api<{ ok: boolean; latencyMs: number; reason: string | null; newStatus: ProviderStatus }>(
   365→        "/api/settings/providers", { method: "POST", body: JSON.stringify({ providerId: p.providerId }) }
   366→      );
   367→      setProviders((prev) => (prev ?? []).map((x) => (x.providerId === p.providerId ? { ...x, status: res.newStatus, lastHealthAt: new Date().toISOString(), lastHealthOk: res.ok } : x)));
   368→      if (res.ok) toast.success(t("settings.healthOk", { ms: res.latencyMs }));
   369→      else toast.warning(t("settings.healthFail", { reason: res.reason ?? "unknown" }));
   370→    } catch (e) {
   371→      toast.error(t("settings.healthFail", { reason: (e as Error).message }));
   372→    } finally {
   373→      setChecking(null);
   374→    }
   375→  };
   376→
   377→  const grouped = useMemo(() => {
   378→    const map = new Map<string, Provider[]>();
   379→    for (const p of providers ?? []) {
   380→      const arr = map.get(p.category) ?? [];
   381→      arr.push(p);
   382→      map.set(p.category, arr);
   383→    }
   384→    const cats = [...map.keys()].sort((a, b) => {
   385→      const ia = CATEGORY_ORDER.indexOf(a), ib = CATEGORY_ORDER.indexOf(b);
   386→      return (ia === -1 ? 99 : ia) - (ib === -1 ? 99 : ib);
   387→    });
   388→    return cats.map((c) => ({ category: c, providers: map.get(c) ?? [] }));
   389→  }, [providers]);
   390→
   391→  return (
   392→    <div className="grid gap-4">
   393→      <div className="flex flex-wrap items-center justify-between gap-2">
   394→        <p className="text-sm text-muted-foreground">{t("settings.providersDesc")}</p>
   395→        <Button variant="outline" size="sm" className="min-h-11" onClick={load} aria-label={t("content.refresh")}>
   396→          <RefreshCw className="h-4 w-4" aria-hidden />
   397→        </Button>
   398→      </div>
   399→
   400→      {!providers ? (
   401→        <div className="grid gap-3">{[...Array(6)].map((_, i) => <Skeleton key={i} className="h-20 rounded-2xl bg-muted/40" />)}</div>
   402→      ) : (
   403→        grouped.map(({ category, providers: list }) => (
   404→          <Card key={category} className="glass rounded-2xl">
   405→            <CardHeader className="pb-2">
   406→              <CardTitle className="text-sm font-medium uppercase tracking-wide text-muted-foreground">{t(`settings.cat.${category}`)}</CardTitle>
   407→            </CardHeader>
   408→            <CardContent className="grid gap-2">
   409→              {list.map((p) => {
   410→                const accent = statusBadge(p.status);
   411→                const row = (
   412→                  <div key={p.providerId} className="flex flex-col gap-3 rounded-xl border border-border/60 bg-muted/20 p-3 sm:flex-row sm:items-center">
   413→                    <div className="min-w-0 flex-1">
   414→                      <div className="flex flex-wrap items-center gap-2">
   415→                        <p className="text-sm font-medium">{p.title}</p>
   416→                        <TooltipProvider delayDuration={100}>
   417→                          <Tooltip>
   418→                            <TooltipTrigger asChild>
   419→                              <Badge variant="outline" className={`cursor-help gap-1 text-[10px] ${accent.className ?? ""}`} style={accent.style}>
   420→                                {statusIcon(p.status)} {t(`settings.status.${p.status}`)}
   421→                              </Badge>

   860→// ===== Budget tab =====
   861→function BudgetTab() {
   862→  const { t } = useI18n();
   863→  const [usage, setUsage] = useState<Usage | null>(null);
   864→  const [limits, setLimits] = useState<{ dailyBudget: number; weeklyBudget: number; monthlyBudget: number } | null>(null);
   865→  const [reloadKey, setReloadKey] = useState(0);
   866→
   867→  useEffect(() => {
   868→    let cancelled = false;
   869→    Promise.all([
   870→      api<Usage>("/api/costs"),
   871→      api<{ policy: Pick<Policy, "dailyBudget" | "weeklyBudget" | "monthlyBudget"> }>("/api/autopilot"),
   872→    ])
   873→      .then(([u, ap]) => { if (!cancelled) { setUsage(u); setLimits(ap.policy); } })
   874→      .catch(() => { if (!cancelled) setUsage(null); });
   875→    return () => { cancelled = true; };
   876→  }, [reloadKey]);
   877→
   878→  if (!usage) {
   879→    return <div className="grid gap-3">{[...Array(4)].map((_, i) => <Skeleton key={i} className="h-20 rounded-2xl bg-muted/40" />)}</div>;
   880→  }
   881→
   882→  const periods: { key: "today" | "week" | "month" | "total"; label: string; value: number; limit: number }[] = [
   883→    { key: "today", label: t("settings.budget.today"), value: usage.today, limit: limits?.dailyBudget ?? 0 },
   884→    { key: "week", label: t("settings.budget.week"), value: usage.week, limit: limits?.weeklyBudget ?? 0 },
   885→    { key: "month", label: t("settings.budget.month"), value: usage.month, limit: limits?.monthlyBudget ?? 0 },
   886→    { key: "total", label: t("settings.budget.total"), value: usage.total, limit: usage.total },
   887→  ];
   888→  const maxProvider = Math.max(0.0001, ...(usage.byProvider ?? []).map((p) => p.cost));
   889→  const maxCapability = Math.max(0.0001, ...(usage.byCapability ?? []).map((p) => p.cost));
   890→
   891→  return (
   892→    <div className="grid gap-4">
   893→      <div className="flex flex-wrap items-center justify-between gap-2">
   894→        <p className="text-sm text-muted-foreground">{t("settings.budget.desc")}</p>
   895→        <Button variant="outline" size="sm" className="min-h-11" onClick={() => setReloadKey((k) => k + 1)} aria-label={t("content.refresh")}>
   896→          <RefreshCw className="h-4 w-4" aria-hidden />
   897→        </Button>
   898→      </div>
   899→
   900→      {/* per-period bars */}
   901→      <div className="grid gap-3 sm:grid-cols-2 xl:grid-cols-4">
   902→        {periods.map((p) => {
   903→          const pct = p.limit > 0 ? Math.min(100, (p.value / p.limit) * 100) : 100;
   904→          return (
   905→            <Card key={p.key} className="glass rounded-2xl">
   906→              <CardContent className="p-4">
   907→                <p className="text-xs text-muted-foreground">{p.label}</p>
   908→                <p className="mt-1 text-xl font-semibold tabular-nums neon-text">${p.value.toFixed(3)}</p>
   909→                <Progress value={pct} className="mt-2 h-1.5" aria-label={`${p.label}: $${p.value.toFixed(3)}`} />
   910→                {p.key !== "total" && (
   911→                  <p className="mt-1.5 text-[11px] text-muted-foreground">
   912→                    {p.limit > 0 ? t("settings.budget.limitLabel", { v: p.limit.toFixed(2) }) : "—"}
   913→                  </p>
   914→                )}
   915→              </CardContent>
   916→            </Card>
   917→          );
   918→        })}
   919→      </div>
   920→
   921→      {/* breakdowns */}
   922→      {usage.total === 0 ? (
   923→        <Card className="glass rounded-2xl"><CardContent className="p-6 text-center text-sm text-muted-foreground">{t("settings.budget.noUsage")}</CardContent></Card>
   924→      ) : (
   925→        <div className="grid gap-4 lg:grid-cols-2">
   926→          <Card className="glass rounded-2xl">
   927→            <CardHeader className="pb-2">
   928→              <CardTitle className="text-sm font-medium uppercase tracking-wide text-muted-foreground">{t("settings.budget.byProvider")}</CardTitle>
   929→            </CardHeader>
   930→            <CardContent className="grid gap-3">
   931→              {(usage.byProvider ?? []).map((p) => (
   932→                <div key={p.provider}>
   933→                  <div className="mb-1 flex justify-between text-xs">
   934→                    <span className="font-mono">{p.provider}</span>
   935→                    <span className="font-medium tabular-nums">${p.cost.toFixed(4)}</span>
   936→                  </div>
   937→                  <Progress value={(p.cost / maxProvider) * 100} className="h-1.5" aria-label={p.provider} />
   938→                </div>
   939→              ))}
   940→            </CardContent>
   941→          </Card>
   942→          <Card className="glass rounded-2xl">
   943→            <CardHeader className="pb-2">
   944→              <CardTitle className="flex items-center gap-2 text-sm font-medium uppercase tracking-wide text-muted-foreground">
   945→                <Eye className="h-3.5 w-3.5" aria-hidden /> {t("settings.budget.byCapability")}
   946→              </CardTitle>
   947→            </CardHeader>
   948→            <CardContent className="grid gap-3">
   949→              {(usage.byCapability ?? []).map((p) => (
   950→                <div key={p.capability}>
   951→                  <div className="mb-1 flex justify-between text-xs">
   952→                    <span className="font-mono">{p.capability}</span>
   953→                    <span className="font-medium tabular-nums">${p.cost.toFixed(4)}</span>
   954→                  </div>
   955→                  <Progress value={(p.cost / maxCapability) * 100} className="h-1.5" aria-label={p.capability} />
   956→                </div>

   958→            </CardContent>
