     1→"use client";
     2→
     3→import { useCallback, useEffect, useState } from "react";
     4→import { motion } from "framer-motion";
     5→import { toast } from "sonner";
     6→import { useApp, pulseCore } from "@/lib/store";
     7→import { useI18n, api } from "@/lib/use-i18n";
     8→import { Button } from "@/components/ui/button";
     9→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
    10→import { Badge } from "@/components/ui/badge";
    11→import { Skeleton } from "@/components/ui/skeleton";
    12→import { Input } from "@/components/ui/input";
    13→import { Label } from "@/components/ui/label";
    14→import { Progress } from "@/components/ui/progress";
    15→import {
    16→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    17→} from "@/components/ui/select";
    18→import {
    19→  Flame, Search, Loader2, ExternalLink, Sparkles, ShieldAlert, Copy, Radar,
    20→  ChevronDown, ChevronUp, Gauge, Globe2, Activity, FilePlus2, Quote,
    21→} from "lucide-react";
    22→
    23→// ---------- types ----------
    24→interface EvidenceQuote { quote: string; source: string; observedAt: string }
    25→interface TrendSource { name: string; url: string; snippet: string; host: string; date: string | null }
    26→interface TrendDto {
    27→  id: string;
    28→  title: string;
    29→  summary: string | null;
    30→  sourceName: string;
    31→  sourceUrl: string;
    32→  platform: string | null;
    33→  region: string | null;
    34→  language: string | null;
    35→  status: string; // VERIFIED_TREND | POPULAR_TOPIC | EMERGING_SIGNAL | HYPOTHESIS
    36→  confidence: number;
    37→  brandFitScore: number | null;
    38→  brandFitReason: string | null;
    39→  suggestedAdaptation: string | null;
    40→  risk: string | null;
    41→  concept: string | null;
    42→  hook: string | null;
    43→  script: string | null;
    44→  evidence?: EvidenceQuote[];
    45→  metricsObserved?: string | null;
    46→  createdAt: string;
    47→}
    48→interface BrandLite { id: string; name: string }
    49→
    50→function errMessage(e: unknown): string {
    51→  return e instanceof Error ? e.message : String(e);
    52→}
    53→
    54→const STATUS_STYLE: Record<string, { color: string }> = {
    55→  VERIFIED_TREND: { color: "var(--neon-2)" },
    56→  POPULAR_TOPIC: { color: "var(--neon-3)" },
    57→  EMERGING_SIGNAL: { color: "var(--neon)" },
    58→  HYPOTHESIS: { color: "var(--muted-foreground)" },
    59→};
    60→const STATUSES = ["VERIFIED_TREND", "POPULAR_TOPIC", "EMERGING_SIGNAL", "HYPOTHESIS"] as const;
    61→
    62→export function TrendsModule() {
    63→  const { t, locale, locales, localeLabels } = useI18n();
    64→  const activeBrandId = useApp((s) => s.activeBrandId);
    65→  const setView = useApp((s) => s.setView);
    66→  const setContentSeed = useApp((s) => s.setContentSeed);
    67→
    68→  const [brands, setBrands] = useState<BrandLite[]>([]);
    69→  // null = no explicit choice yet → derive from the active brand (effect-free, always in sync)
    70→  const [pickedBrandId, setPickedBrandId] = useState<string | null>(null);
    71→  const brandId = pickedBrandId ?? activeBrandId ?? "none";
    72→  const [niche, setNiche] = useState("");
    73→  const [region, setRegion] = useState("");
    74→  const [language, setLanguage] = useState<string>(locale);
    75→
    76→  const [trends, setTrends] = useState<TrendDto[]>([]);
    77→  const [loading, setLoading] = useState(true);
    78→  const [searching, setSearching] = useState(false);
    79→  const [adaptingId, setAdaptingId] = useState<string | null>(null);
    80→  const [draftingId, setDraftingId] = useState<string | null>(null);
    81→  const [expanded, setExpanded] = useState<Record<string, boolean>>({});
    82→  // real web-search results backing the latest Trend Agent run
    83→  const [sources, setSources] = useState<TrendSource[]>([]);
    84→  const [sourcesOpen, setSourcesOpen] = useState(false);
    85→  // honest degradation: live search unavailable → LLM hypotheses (unverified)
    86→  const [searchFallback, setSearchFallback] = useState(false);
    87→
    88→  // keep trend language in sync with UI locale
    89→  useEffect(() => {
    90→    setLanguage(locale);
    91→  }, [locale]);
    92→
    93→  const loadInitial = useCallback(async () => {
    94→    try {
    95→      const [trendList, brandList] = await Promise.all([
    96→        api<TrendDto[]>("/api/trends"),
    97→        api<BrandLite[]>("/api/brands"),
    98→      ]);
    99→      setTrends(trendList);
   100→      setBrands(brandList);
   101→    } catch (e) {
   102→      toast.error(errMessage(e));
   103→    } finally {
   104→      setLoading(false);
   105→    }
   106→  }, []);
   107→
   108→  useEffect(() => {
   109→    loadInitial();
   110→  }, [loadInitial]);
   111→
   112→  const search = async () => {
   113→    setSearching(true);
   114→    pulseCore("TREND_SEARCH");
   115→    try {
   116→      const res = await api<{ jobId: string; trends: TrendDto[]; sources: TrendSource[]; sourcesCount: number; searchFallback?: boolean }>("/api/trends", {
   117→        method: "POST",
   118→        body: JSON.stringify({
   119→          brandId: brandId === "none" ? undefined : brandId,
   120→          niche: niche.trim() || undefined,
   121→          region: region.trim() || undefined,
   122→          language,
   123→        }),
   124→      });
   125→      pulseCore("SUCCESS");
   126→      toast.success(t("trends.found", { count: res.trends.length }));
   127→      setTrends((prev) => [...res.trends, ...prev]);
   128→      setSources(res.sources ?? []);
   129→      setSourcesOpen(true);
   130→      setSearchFallback(Boolean(res.searchFallback));
   131→    } catch (e) {
   132→      pulseCore("ERROR");
   133→      toast.error(errMessage(e));
   134→    } finally {
   135→      setSearching(false);
   136→    }
   137→  };
   138→
   139→  // Trend → Content bridge: AI writes a platform-native draft grounded in the trend adaptation,
   140→  // then Content auto-opens the new draft via the transient contentSeed
   141→  const draftFromTrend = async (tr: TrendDto) => {
   142→    const targetBrandId = brandId !== "none" ? brandId : activeBrandId ?? brands[0]?.id;
   143→    if (!targetBrandId) {
   144→      toast.error(t("trends.needBrand"));
   145→      return;
   146→    }
   147→    setDraftingId(tr.id);
   148→    pulseCore("GENERATING");
   149→    try {
   150→      const item = await api<{ id: string }>("/api/content", {
   151→        method: "POST",
   152→        body: JSON.stringify({
   153→          brandId: targetBrandId,
   154→          title: tr.title,
   155→          platform: tr.platform && tr.platform !== "web" ? tr.platform : "instagram",
   156→          language,
   157→          contentType: tr.platform === "tiktok" ? "VIDEO_REEL" : "IMAGE_POST",
   158→          aiWrite: true,
   159→          brief: tr.suggestedAdaptation ?? tr.summary ?? tr.title,
   160→          linkTrend: tr.sourceUrl,
   161→          trendId: tr.id,
   162→          meta: { fromTrend: tr.id, fromTrendSource: tr.sourceUrl },
   163→        }),
   164→      });
   165→      pulseCore("SUCCESS");
   166→      toast.success(t("trends.toDraftDone"));
   167→      setContentSeed(item.id);
   168→      setView("content");
   169→    } catch (e) {
   170→      pulseCore("ERROR");
   171→      toast.error(errMessage(e));
   172→    } finally {
   173→      setDraftingId(null);
   174→    }
   175→  };
   176→
   177→  const adapt = async (trend: TrendDto) => {
   178→    setAdaptingId(trend.id);
   179→    pulseCore("GENERATING");
   180→    try {
   181→      const updated = await api<TrendDto>(`/api/trends/${trend.id}`, {
   182→        method: "PATCH",
   183→        body: JSON.stringify({ action: "adapt", language }),
   184→      });
   185→      setTrends((prev) => prev.map((x) => (x.id === updated.id ? { ...x, ...updated } : x)));
   186→      setExpanded((prev) => ({ ...prev, [trend.id]: true }));
   187→      pulseCore("SUCCESS");
   188→      toast.success(t("trends.adapted"));
   189→    } catch (e) {
   190→      pulseCore("ERROR");
   191→      toast.error(errMessage(e));
   192→    } finally {
   193→      setAdaptingId(null);
   194→    }
   195→  };
   196→
   197→  const copyText = async (text: string) => {
   198→    try {
   199→      await navigator.clipboard.writeText(text);
   200→      toast.success(t("common.copied"));
   201→    } catch {
   202→      toast.error(t("common.error"));
   203→    }
   204→  };
   205→
   206→  const hasAdaptation = (tr: TrendDto) => Boolean(tr.concept || tr.hook || tr.script);
   207→  const hostOf = (url: string) => {
   208→    try { return new URL(url).host; } catch { return url; }
   209→  };
   210→
   211→  return (
   212→    <div className="grid gap-5">
   213→      {/* header */}
   214→      <motion.section initial={{ opacity: 0, y: 12 }} animate={{ opacity: 1, y: 0 }}>
   215→        <h1 className="text-xl font-semibold sm:text-2xl">{t("trends.title")}</h1>
   216→        <p className="mt-1 text-sm text-muted-foreground">{t("trends.subtitle")}</p>
   217→      </motion.section>
   218→
   219→      {/* search form */}
   220→      <motion.section initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} aria-label={t("trends.searchTitle")}>
   221→        <Card className="glass rounded-2xl">
   222→          <CardHeader className="pb-2">
   223→            <CardTitle className="flex items-center gap-2 text-base">
   224→              <Radar className="h-4 w-4 text-[var(--neon-3)]" /> {t("trends.searchTitle")}
   225→            </CardTitle>
   226→          </CardHeader>
   227→          <CardContent className="grid gap-4">
   228→            <form
   229→              className="grid gap-3 sm:grid-cols-2 lg:grid-cols-4"
   230→              onSubmit={(e) => { e.preventDefault(); if (!searching) search(); }}
   231→            >
   232→              <div className="grid gap-1.5">
   233→                <Label htmlFor="tr-brand">{t("trends.brand")}</Label>
   234→                <Select value={brandId} onValueChange={setPickedBrandId}>
   235→                  <SelectTrigger id="tr-brand" className="h-11">
   236→                    <SelectValue placeholder={t("trends.brandAny")} />
   237→                  </SelectTrigger>
   238→                  <SelectContent>
   239→                    <SelectItem value="none">{t("trends.brandAny")}</SelectItem>
   240→                    {brands.map((b) => (
   241→                      <SelectItem key={b.id} value={b.id}>{b.name}</SelectItem>
   242→                    ))}
   243→                  </SelectContent>
   244→                </Select>
   245→              </div>
   246→              <div className="grid gap-1.5">
   247→                <Label htmlFor="tr-niche">{t("trends.niche")}</Label>
   248→                <Input
   249→                  id="tr-niche"
   250→                  value={niche}
   251→                  onChange={(e) => setNiche(e.target.value)}
   252→                  placeholder={t("trends.nichePlaceholder")}
   253→                  className="h-11"
   254→                />
   255→              </div>
   256→              <div className="grid gap-1.5">
   257→                <Label htmlFor="tr-region">{t("trends.region")}</Label>
   258→                <Input
   259→                  id="tr-region"
   260→                  value={region}
   261→                  onChange={(e) => setRegion(e.target.value)}
   262→                  placeholder={t("trends.regionPlaceholder")}
   263→                  className="h-11"
   264→                />
   265→              </div>
   266→              <div className="grid gap-1.5">
   267→                <Label htmlFor="tr-lang">{t("trends.language")}</Label>
   268→                <Select value={language} onValueChange={setLanguage}>
   269→                  <SelectTrigger id="tr-lang" className="h-11">
   270→                    <SelectValue />
   271→                  </SelectTrigger>
   272→                  <SelectContent>
   273→                    {locales.map((l) => (
   274→                      <SelectItem key={l} value={l}>{localeLabels[l]}</SelectItem>
   275→                    ))}
   276→                  </SelectContent>
   277→                </Select>
   278→              </div>
   279→              <div className="flex items-end gap-3 sm:col-span-2 lg:col-span-4">
   280→                <Button type="submit" className="h-11 gap-2 neon-border" disabled={searching} aria-label={t("trends.search")}>
   281→                  {searching ? <Loader2 className="h-4 w-4 animate-spin" /> : <Search className="h-4 w-4" />}
   282→                  {searching ? t("trends.searching") : t("trends.search")}
   283→                </Button>
   284→                <p className="text-xs text-muted-foreground" role="note">{t("trends.searchHint")}</p>
   285→              </div>
   286→            </form>
   287→          </CardContent>
   288→        </Card>
   289→      </motion.section>
   290→
   291→      {/* real sources backing the latest Trend Agent run */}
   292→      {sources.length > 0 && (
   293→        <motion.section
   294→          initial={{ opacity: 0, y: 10 }}
   295→          animate={{ opacity: 1, y: 0 }}
   296→          aria-label={t("trends.sourcesTitle")}
   297→        >
   298→          <Card className="glass rounded-2xl">
   299→            <CardHeader className="pb-2">
   300→              <div className="flex flex-wrap items-center justify-between gap-2">
   301→                <CardTitle className="flex items-center gap-2 text-base">
   302→                  <span
   303→                    className="rounded-lg p-1.5"
   304→                    style={{ background: "color-mix(in oklab, var(--neon-2) 15%, transparent)" }}
   305→                  >
   306→                    <Globe2 className="h-4 w-4 text-[var(--neon-2)]" />
   307→                  </span>
   308→                  {t("trends.sourcesTitle")}
   309→                  <Badge variant="outline" className="text-[10px]" aria-label={String(sources.length)}>
   310→                    {sources.length}
   311→                  </Badge>
   312→                </CardTitle>
   313→                <Button
   314→                  variant="ghost"
   315→                  size="sm"
   316→                  className="h-8 gap-1.5 text-xs"
   317→                  onClick={() => setSourcesOpen((o) => !o)}
   318→                  aria-expanded={sourcesOpen}
   319→                  aria-label={t("trends.sourcesTitle")}
   320→                >
   321→                  {sourcesOpen ? <ChevronUp className="h-4 w-4" /> : <ChevronDown className="h-4 w-4" />}
   322→                </Button>
   323→              </div>
   324→              <p className="text-xs text-muted-foreground">{t("trends.sourcesHint")}</p>
   325→            </CardHeader>
   326→            {sourcesOpen && (
   327→              <CardContent className="grid gap-2 sm:grid-cols-2 xl:grid-cols-3">
   328→                {sources.map((s, i) => (
   329→                  <motion.a
   330→                    key={`${s.url}-${i}`}
   331→                    href={s.url}
   332→                    target="_blank"
   333→                    rel="noopener noreferrer"
   334→                    initial={{ opacity: 0, y: 6 }}
   335→                    animate={{ opacity: 1, y: 0 }}
   336→                    transition={{ delay: Math.min(i * 0.04, 0.25) }}
   337→                    className="source-card group/source"
   338→                    aria-label={`${s.host || "web"}: ${s.name}`}
   339→                  >
   340→                    <div className="flex items-center gap-2">
   341→                      <span className="rank-chip" aria-hidden="true">{String(i + 1).padStart(2, "0")}</span>
   342→                      <span className="min-w-0 flex-1 truncate text-xs font-semibold">{s.host || s.name}</span>
   343→                      {s.date && <span className="shrink-0 text-[10px] text-muted-foreground">{s.date}</span>}
   344→                      <ExternalLink className="h-3 w-3 shrink-0 text-muted-foreground opacity-0 transition-opacity group-hover/source:opacity-100" aria-hidden="true" />
   345→                    </div>
   346→                    <p className="mt-1.5 line-clamp-2 text-[11px] leading-snug text-muted-foreground">{s.name}</p>
   347→                  </motion.a>
   348→                ))}
   349→              </CardContent>
   350→            )}
   351→          </Card>
   352→        </motion.section>
   353→      )}
   354→
   355→      {/* trend feed */}
   356→      <section aria-label={t("trends.feed")} className="grid gap-4">
   357→        <h2 className="flex items-center gap-2 text-sm font-medium tracking-wide text-muted-foreground">
   358→          <Flame className="h-4 w-4 text-[var(--neon)]" /> {t("trends.feed")}
   359→        </h2>
   360→
   361→        {searchFallback && (
   362→          <div
   363→            role="status"
   364→            className="flex items-start gap-2 rounded-xl border p-3 text-[13px] leading-relaxed"
   365→            style={{
   366→              borderColor: "color-mix(in oklab, var(--neon-3) 35%, transparent)",
   367→              background: "color-mix(in oklab, var(--neon-3) 8%, transparent)",
   368→            }}
   369→          >
   370→            <ShieldAlert className="mt-0.5 h-4 w-4 shrink-0 text-[var(--neon-3)]" aria-hidden="true" />
   371→            <span>{t("trends.fallbackNotice")}</span>
   372→          </div>
   373→        )}
   374→
   375→        {loading ? (
   376→          [...Array(3)].map((_, i) => <Skeleton key={i} className="h-44 rounded-2xl bg-muted/40" />)
   377→        ) : trends.length === 0 ? (
   378→          <Card className="glass rounded-2xl">
   379→            <CardContent className="flex min-h-40 flex-col items-center justify-center gap-3 p-6 text-center">
   380→              <span className="rounded-2xl p-3" style={{ background: "color-mix(in oklab, var(--neon-3) 15%, transparent)" }}>
   381→                <Flame className="h-7 w-7 text-[var(--neon-3)]" />
   382→              </span>
   383→              <p className="text-sm text-muted-foreground">{t("trends.noTrends")}</p>
   384→            </CardContent>
   385→          </Card>
   386→        ) : (
   387→          trends.map((tr, i) => {
   388→            const statusStyle = STATUS_STYLE[tr.status] ?? STATUS_STYLE.HYPOTHESIS;
   389→            const fitPct = Math.round((tr.brandFitScore ?? 0) * 100);
   390→            const adapted = hasAdaptation(tr);
   391→            const isOpen = Boolean(expanded[tr.id]);
   392→            return (
   393→              <motion.article
   394→                key={tr.id}
   395→                initial={{ opacity: 0, y: 10 }}
   396→                animate={{ opacity: 1, y: 0 }}
   397→                transition={{ delay: Math.min(i * 0.03, 0.2) }}
   398→              >
   399→                <Card className="glass glass-hover rounded-2xl">
   400→                  <CardContent className="grid gap-3 p-4 sm:p-5">
   401→                    <div className="flex flex-wrap items-start justify-between gap-2">
   402→                      <h3 className="min-w-0 flex-1 text-sm font-semibold leading-snug sm:text-base">{tr.title}</h3>
   403→                      <Badge
   404→                        variant="outline"
   405→                        className="shrink-0 text-[10px]"
   406→                        style={{
   407→                          color: statusStyle.color,
   408→                          borderColor: `color-mix(in oklab, ${statusStyle.color} 45%, transparent)`,
   409→                          background: `color-mix(in oklab, ${statusStyle.color} 12%, transparent)`,
   410→                        }}
   411→                      >
   412→                        {t(`trends.status.${STATUSES.includes(tr.status as (typeof STATUSES)[number]) ? tr.status : "HYPOTHESIS"}` as const)}
   413→                      </Badge>
   414→                    </div>
   415→
   416→                    {tr.summary && (
   417→                      <p className="line-clamp-3 text-sm leading-relaxed text-muted-foreground">{tr.summary}</p>
   418→                    )}
   419→
   420→                    <div className="flex flex-wrap items-center gap-2 text-[11px] text-muted-foreground">
   421→                      {tr.platform && (
   422→                        <span className="rounded-md border border-border/60 bg-muted/30 px-2 py-0.5 font-medium">{tr.platform}</span>
   423→                      )}
   424→                      {tr.region && <span>{tr.region}</span>}
   425→                      <span className="inline-flex items-center gap-1">
   426→                        <Gauge className="h-3 w-3" /> {t("trends.confidence")}: {Math.round(tr.confidence * 100)}%
   427→                      </span>
   428→                      {tr.metricsObserved && (
   429→                        <span className="inline-flex max-w-full items-center gap-1 rounded-md border border-border/60 bg-muted/30 px-2 py-0.5" title={tr.metricsObserved}>
   430→                          <Activity className="h-3 w-3 shrink-0 text-[var(--neon-3)]" aria-hidden="true" />
   431→                          <span className="max-w-48 truncate font-medium text-[var(--neon-3)]">{tr.metricsObserved}</span>
   432→                        </span>
   433→                      )}
   434→                    </div>
   435→
   436→                    {tr.evidence && tr.evidence.length > 0 && (
   437→                      <blockquote className="quote-accent">
   438→                        <p className="line-clamp-2 text-[13px] leading-relaxed">{tr.evidence[0].quote}</p>
   439→                        {tr.evidence[0].source && (
   440→                          <a
   441→                            href={tr.evidence[0].source}
   442→                            target="_blank"
   443→                            rel="noopener noreferrer"
   444→                            className="mt-1 inline-flex items-center gap-1 text-[11px] text-muted-foreground underline-offset-2 hover:text-[var(--neon-2)] hover:underline"
   445→                          >
   446→                            <ExternalLink className="h-3 w-3" aria-hidden="true" />
   447→                            {hostOf(tr.evidence[0].source)}
   448→                          </a>
   449→                        )}
   450→                      </blockquote>
   451→                    )}
   452→
   453→                    {tr.brandFitScore != null && (
   454→                      <div className="grid gap-1.5">
   455→                        <div className="flex items-center justify-between text-[11px]">
   456→                          <span className="text-muted-foreground">{t("trends.brandFit")}</span>
   457→                          <span className="font-semibold text-[var(--neon-2)]">{fitPct}%</span>
   458→                        </div>
   459→                        <Progress value={fitPct} className="h-1.5 [&>div]:bg-[var(--neon-2)]" aria-label={t("trends.brandFit")} />
   460→                        {tr.brandFitReason && (
   461→                          <p className="text-[11px] leading-snug text-muted-foreground">
   462→                            <span className="font-medium text-[var(--neon-2)]">{t("trends.why")}:</span> {tr.brandFitReason}
   463→                          </p>
   464→                        )}
   465→                      </div>
   466→                    )}
   467→
   468→                    {tr.suggestedAdaptation && (
   469→                      <div
   470→                        className="rounded-xl border p-3 text-[13px] leading-relaxed"
   471→                        style={{
   472→                          borderColor: "color-mix(in oklab, var(--neon) 30%, transparent)",
   473→                          background: "color-mix(in oklab, var(--neon) 7%, transparent)",
   474→                        }}
   475→                      >
   476→                        <p className="mb-1 flex items-center gap-1.5 text-[11px] font-semibold uppercase tracking-wide text-[var(--neon)]">
   477→                          <Sparkles className="h-3.5 w-3.5" /> {t("trends.adaptation")}
   478→                        </p>
   479→                        {tr.suggestedAdaptation}
   480→                      </div>
   481→                    )}
   482→
   483→                    {tr.risk && (
