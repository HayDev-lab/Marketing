    79→// Honest color-coding: emerald=approved flow, amber=attention, red=problem, fuchsia=in-flight
    80→function stateAccent(s: ApprovalState): { style: React.CSSProperties; className?: string } {
    81→  switch (s) {
    82→    case "APPROVED":
    83→      return { style: { background: "color-mix(in oklab, var(--neon-2) 16%, transparent)", color: "var(--neon-2)", borderColor: "color-mix(in oklab, var(--neon-2) 40%, transparent)" } };
    84→    case "PUBLISHED":
    85→      return { style: { background: "color-mix(in oklab, var(--neon-2) 26%, transparent)", color: "var(--neon-2)", borderColor: "color-mix(in oklab, var(--neon-2) 55%, transparent)" } };
    86→    case "SCHEDULED":
    87→      return { style: { background: "color-mix(in oklab, var(--neon) 16%, transparent)", color: "var(--neon)", borderColor: "color-mix(in oklab, var(--neon) 40%, transparent)" } };
    88→    case "GENERATING":
    89→      return { style: { background: "color-mix(in oklab, var(--neon-3) 14%, transparent)", color: "var(--neon-3)", borderColor: "color-mix(in oklab, var(--neon-3) 40%, transparent)" } };
    90→    case "READY_FOR_REVIEW":
    91→      return { style: { background: "color-mix(in oklab, var(--neon-3) 22%, transparent)", color: "var(--neon-3)", borderColor: "color-mix(in oklab, var(--neon-3) 55%, transparent)" } };
    92→    case "CHANGES_REQUESTED":
    93→    case "FAILED":
    94→      return { style: {}, className: "bg-destructive/15 text-destructive border-destructive/40" };
    95→    default:
    96→      return { style: {}, className: "bg-muted/40 text-muted-foreground border-border" };

   126→// ---------- content package export ----------
   127→function buildMarkdown(d: ContentDetail, brandName: string): string {
   128→  const assetUrl = d.asset ? `${window.location.origin}/api/assets/${d.asset.id}/raw` : null;
   129→  const lines: string[] = [
   130→    `# ${d.title}`,
   131→    "",
   132→    `- **Brand:** ${brandName || "—"}`,
   133→    `- **Platform:** ${d.platform}`,
   134→    `- **Type:** ${d.contentType}`,
   135→    `- **Language:** ${d.language}`,
   136→    `- **State:** ${d.approvalState}`,
   137→    `- **Version:** v${d.contentVersion}`,
   138→    `- **Updated:** ${new Date(d.updatedAt).toISOString()}`,
   139→    "",
   140→  ];
   141→  if (d.hook) lines.push("## Hook", "", d.hook, "");
   142→  if (d.caption) lines.push("## Caption", "", d.caption, "");
   143→  if (d.hashtags) lines.push("## Hashtags", "", d.hashtags, "");
   144→  if (d.script) lines.push("## Script", "", d.script, "");
   145→  if (d.videoProject?.scenes?.length) {
   146→    lines.push("## Video scenes", "");
   147→    d.videoProject.scenes
   148→      .slice()
   149→      .sort((a, b) => a.order - b.order)
   150→      .forEach((s, i) => lines.push(`${i + 1}. ${s.text ?? s.narration ?? ""}`));
   151→    lines.push("");
   152→  }
   153→  if (assetUrl) lines.push(`## Media`, "", `${d.asset?.filename ?? "asset"} (${d.asset?.mimeType ?? ""}) — ${assetUrl}`, "");
   154→  lines.push("---", "", `Exported from ՀայDev Marketing — ${new Date().toISOString()}`);
   155→  return lines.join("\n");
   156→}

   158→function downloadText(filename: string, text: string, mime: string) {
   159→  const blob = new Blob([text], { type: `${mime};charset=utf-8` });
   160→  const url = URL.createObjectURL(blob);
   161→  const a = document.createElement("a");
   162→  a.href = url;
   163→  a.download = filename;
   164→  document.body.appendChild(a);
   165→  a.click();
   166→  a.remove();
   167→  setTimeout(() => URL.revokeObjectURL(url), 1500);
   168→}

   170→function safeFileStem(s: string): string {
   171→  return (s || "content").toLowerCase().replace(/[^a-z0-9\u0561-\u0587\u0430-\u044f]+/gi, "-").replace(/^-+|-+$/g, "").slice(0, 48) || "content";

   450→              <div className="flex flex-wrap items-center gap-2">
   451→                <StateBadge state={detail.approvalState} />
   452→                <Badge variant="outline" className="gap-1 text-[10px]"><PlatformIcon platform={detail.platform} /> {detail.platform}</Badge>
   453→                <Badge variant="outline" className="text-[10px]">{t(`content.lang.${detail.language}`)}</Badge>

   457→                </span>
   458→              </div>
   459→
   460→              {/* invalidation notice (approval honestly invalidated) */}
   461→              {invalidation && (
   462→                <div className="rounded-xl border border-[color-mix(in_oklab,var(--neon-3)_50%,transparent)] bg-[color-mix(in_oklab,var(--neon-3)_10%,transparent)] p-3 text-xs text-foreground" role="alert">
   463→                  <CircleAlert className="mr-1 inline h-4 w-4 text-[var(--neon-3)]" aria-hidden />
   464→                  {invalidation}
   465→                </div>
   466→              )}
   467→
   468→              {/* state-specific honest notes */}
   469→              {detail.approvalState === "APPROVED" && (
   470→                <div className="rounded-xl border border-[color-mix(in_oklab,var(--neon-2)_50%,transparent)] bg-[color-mix(in_oklab,var(--neon-2)_10%,transparent)] p-3 text-xs" role="status">
   471→                  <CheckCircle2 className="mr-1 inline h-4 w-4 text-[var(--neon-2)]" aria-hidden /> {t("content.approvedNote")}
   472→                </div>
   473→              )}
   474→              {detail.approvalState === "GENERATING" && (
   475→                <div className="rounded-xl border border-border/60 bg-muted/20 p-3 text-xs text-muted-foreground" role="status">
   476→                  <Loader2 className="mr-1 inline h-4 w-4 animate-spin" aria-hidden /> {t("content.generatingNote")}
   477→                </div>
   478→              )}
   479→
   480→              {/* approval actions */}
   481→              <section aria-label={t("content.approvalActions")}>
   482→                <h3 className="mb-2 text-xs font-medium uppercase tracking-wide text-muted-foreground">{t("content.approvalActions")}</h3>
   483→                <div className="flex flex-wrap gap-2">
   484→                  {(detail.approvalState === "DRAFT" || detail.approvalState === "FAILED") && (
   485→                    <Button size="sm" className="min-h-11" disabled={transitioning} onClick={() => transition("READY_FOR_REVIEW")}>
   486→                      {transitioning ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <ShieldCheck className="h-4 w-4" aria-hidden />}
   487→                      {t("content.submitReview")}
   488→                    </Button>
   489→                  )}
   490→                  {detail.approvalState === "READY_FOR_REVIEW" && (
   491→                    <>
   492→                      <Button size="sm" className="min-h-11" disabled={transitioning} onClick={() => transition("APPROVED")}>
   493→                        {transitioning ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <CheckCircle2 className="h-4 w-4" aria-hidden />}
   494→                        {t("content.approve")}
   495→                      </Button>
   496→                      <Button size="sm" variant="outline" className="min-h-11" disabled={transitioning} onClick={() => transition("CHANGES_REQUESTED")}>
   497→                        {t("content.requestChanges")}
   498→                      </Button>
   499→                      <Button size="sm" variant="ghost" className="min-h-11" disabled={transitioning} onClick={() => transition("DRAFT")}>
   500→                        {t("content.backToDraft")}
   501→                      </Button>
   502→                    </>
   503→                  )}
   504→                  {detail.approvalState === "CHANGES_REQUESTED" && (
   505→                    <>
   506→                      <Button size="sm" className="min-h-11" disabled={transitioning} onClick={() => transition("READY_FOR_REVIEW")}>
   507→                        {t("content.submitReview")}
   508→                      </Button>
   509→                      <Button size="sm" variant="ghost" className="min-h-11" disabled={transitioning} onClick={() => transition("DRAFT")}>
   510→                        {t("content.backToDraft")}
   511→                      </Button>
   512→                    </>
   513→                  )}
   514→                  {(detail.approvalState === "SCHEDULED" || detail.approvalState === "PUBLISHED") && (
   515→                    <p className="text-xs text-muted-foreground">{t("content.readonlyNote")}</p>
   516→                  )}
   517→                </div>
   518→              </section>
   519→
   520→              {/* media preview */}
   521→              <section aria-label={t("content.media")}>
   522→                <h3 className="mb-2 text-xs font-medium uppercase tracking-wide text-muted-foreground">{t("content.media")}</h3>
   523→                {detail.asset ? (
   524→                  detail.asset.kind === "IMAGE" ? (
   525→                    <img src={`/api/assets/${detail.asset.id}/raw`} alt={detail.asset.filename} className="max-h-72 w-full rounded-xl border border-border/60 object-cover" />
   526→                  ) : detail.asset.kind === "VIDEO" ? (
   527→                    <video src={`/api/assets/${detail.asset.id}/raw`} controls className="max-h-72 w-full rounded-xl border border-border/60" aria-label={detail.asset.filename} />
   528→                  ) : (
   529→                    <div className="flex items-center gap-3 rounded-xl border border-border/60 bg-muted/20 p-3">
   530→                      <AudioLines className="h-5 w-5 text-[var(--neon-2)]" aria-hidden />
   531→                      <audio src={`/api/assets/${detail.asset.id}/raw`} controls className="w-full" aria-label={detail.asset.filename} />
   532→                    </div>
   533→                  )
   534→                ) : detail.videoProject ? (
   535→                  <div className="rounded-xl border border-border/60 bg-muted/20 p-3">
   536→                    <p className="mb-2 flex items-center gap-2 text-xs text-muted-foreground">
   537→                      <Film className="h-4 w-4 text-[var(--neon)]" aria-hidden /> {detail.videoProject.scenes.length} scenes
   538→                    </p>
   539→                    <ul className="max-h-40 space-y-1 overflow-y-auto scrollbar-thin text-xs">
   540→                      {detail.videoProject.scenes.map((sc) => (
   541→                        <li key={sc.id} className="truncate rounded-lg bg-muted/30 px-2 py-1">{sc.text ?? sc.narration ?? "—"}</li>
   542→                      ))}
   543→                    </ul>
   544→                  </div>
   545→                ) : (
   546→                  <p className="flex items-center gap-2 text-xs text-muted-foreground"><ImageIcon className="h-4 w-4" aria-hidden /> {t("content.noMedia")}</p>
   547→                )}
   548→              </section>
   549→
   550→              <Separator />
   551→
   552→              {/* editable fields */}
   553→              <section aria-label={t("content.field.title")} className="grid gap-3">
   554→                <div className="grid gap-1.5">
   555→                  <Label htmlFor="c-title">{t("content.field.title")}</Label>
   556→                  <Input id="c-title" className="min-h-11" value={form.title} disabled={!editable} onChange={(e) => setForm({ ...form, title: e.target.value })} />
   557→                </div>
   558→                <div className="grid gap-1.5">
   559→                  <Label htmlFor="c-hook">{t("content.field.hook")}</Label>
   560→                  <Input id="c-hook" className="min-h-11" value={form.hook} disabled={!editable} onChange={(e) => setForm({ ...form, hook: e.target.value })} />

   562→                <div className="grid gap-1.5">
   563→                  <Label htmlFor="c-caption">{t("content.field.caption")}</Label>

   566→                <div className="grid gap-1.5">
   567→                  <Label htmlFor="c-tags">{t("content.field.hashtags")}</Label>
   568→                  <Input id="c-tags" className="min-h-11" value={form.hashtags} disabled={!editable} onChange={(e) => setForm({ ...form, hashtags: e.target.value })} />
   569→                </div>
   570→                <div className="grid gap-1.5">
   571→                  <Label htmlFor="c-script">{t("content.field.script")}</Label>
   572→                  <Textarea id="c-script" rows={5} value={form.script} disabled={!editable} onChange={(e) => setForm({ ...form, script: e.target.value })} />
   573→                </div>
   574→                <div className="grid grid-cols-1 gap-3 sm:grid-cols-3">
   575→                  <div className="grid gap-1.5">
   576→                    <Label>{t("content.field.platform")}</Label>
   577→                    <Select value={form.platform} disabled={!editable} onValueChange={(v) => setForm({ ...form, platform: v })}>
   578→                      <SelectTrigger className="min-h-11"><SelectValue /></SelectTrigger>
   579→                      <SelectContent>
   580→                        {PLATFORMS.map((p) => <SelectItem key={p} value={p}>{p}</SelectItem>)}
   581→                      </SelectContent>
   582→                    </Select>
   583→                  </div>
   584→                  <div className="grid gap-1.5">
   585→                    <Label>{t("content.field.language")}</Label>
   586→                    <Select value={form.language} disabled={!editable} onValueChange={(v) => setForm({ ...form, language: v })}>
   587→                      <SelectTrigger className="min-h-11"><SelectValue /></SelectTrigger>

   589→                        {LANGS.map((l) => <SelectItem key={l} value={l}>{t(`content.lang.${l}`)}</SelectItem>)}
   590→                      </SelectContent>
   591→                    </Select>
   592→                  </div>

   594→                    <Label>{t("content.field.contentType")}</Label>
   595→                    <Select value={form.contentType} disabled={!editable} onValueChange={(v) => setForm({ ...form, contentType: v })}>
   596→                      <SelectTrigger className="min-h-11"><SelectValue /></SelectTrigger>
   597→                      <SelectContent>
   598→                        {TYPES.map((tp) => <SelectItem key={tp} value={tp}>{t(`content.type.${tp}`)}</SelectItem>)}
   599→                      </SelectContent>
   600→                    </Select>
   601→                  </div>
   602→                </div>
   603→                {editable ? (
   604→                  <Button className="min-h-11 w-full sm:w-auto sm:justify-self-end" disabled={saving} onClick={saveEdits}>
   605→                    {saving ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : null}
   606→                    {t("common.save")}
   607→                  </Button>
   608→                ) : (
   609→                  <p className="text-xs text-muted-foreground">{t("content.readonlyNote")}</p>
   610→                )}
   611→              </section>
   612→
   613→              {/* approval history */}
   614→              <section aria-label={t("content.history")}>
   615→                <h3 className="mb-2 flex items-center gap-2 text-xs font-medium uppercase tracking-wide text-muted-foreground">
   616→                  <History className="h-3.5 w-3.5" aria-hidden /> {t("content.history")}
   617→                </h3>

   619→                  <p className="text-xs text-muted-foreground">{t("content.noHistory")}</p>

   621→                  <ol className="max-h-44 space-y-2 overflow-y-auto pr-1 scrollbar-thin">

   625→                          <span className="font-medium">{ev.fromState} → {ev.toState}</span>

   638→              <Separator />

   660→                        }

   662→                    >
   663→                      <Copy className="mr-2 h-4 w-4" aria-hidden /> {t("content.exportCopy")}

   668→                        toast.success(t("content.exportDone"));
   669→                      }}
   670→                    >
   671→                      <FileDown className="mr-2 h-4 w-4" aria-hidden /> {t("content.exportMd")}

   689→            </div>

   691→        </DialogContent>

   693→
   694→      {/* delete confirm */}

   696→        <AlertDialogContent>
   697→          <AlertDialogHeader>
   698→            <AlertDialogTitle>{t("content.delete")}</AlertDialogTitle>
   699→            <AlertDialogDescription>{t("content.deleteConfirm")}</AlertDialogDescription>
   700→          </AlertDialogHeader>
   701→          <AlertDialogFooter>
   702→            <AlertDialogCancel className="min-h-11">{t("common.cancel")}</AlertDialogCancel>
   703→            <AlertDialogAction className="min-h-11 bg-destructive text-destructive-foreground hover:bg-destructive/90" onClick={removeItem}>
   704→              {t("common.delete")}
   705→            </AlertDialogAction>
   706→          </AlertDialogFooter>
   707→        </AlertDialogContent>
